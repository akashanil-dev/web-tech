#include<stdio.h>
#define size 10
int stack[size];
int top = -1;

void push(int value)
{
    if(top == size - 1)
    {
        printf("Overflow");
    }
    else
    {
        top++;
        stack[top] = value;
        printf("Added\n");
    }
}

void pop()
{
    int pop_digit;
    if(top == -1)
    {
        printf("Empty\n");
    }
    else
    {
        pop_digit = stack[top];
        top--;
        printf("%d removed\n", pop_digit);
    }
}

void peek()
{
    if(top == -1)
    {
        printf("Empty\n");
    }
    else
    {
        printf("%d top\n", stack[top]);
    }
}

void display()
{
    int i;
    if(top == -1)
    {
        printf("Empty\n");
    }
    else
    {
        for(i = top; i >= 0; i--)
        {
            printf("%d ", stack[i]);
        }
    }
}

void main()
{
    int choice, value, limit = 1;
    while(limit != 0)
    {
        printf("\n1.Push\n2.Pop\n3.Peek\n4.Show\n5.Exit\nChoose an operation: ");
        scanf("%d", &choice);
        switch(choice)
        {
            case 1:
                printf("Value: ");
                scanf("%d", &value);
                push(value); break;
            case 2:
                pop(); break;
            case 3:
                peek(); break;
            case 4:
                display(); break;
            case 5:
                limit = 0; break;
        }
    }
}
