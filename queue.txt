#include<stdio.h>
#define size 5
int queue[size], i = 0;
int front, rear = -1;


void enqueue(int value)
{
    if(rear == size - 1)
    {
        printf("Overflow\n");
    }
    else
    {
        if(rear == -1)
        {
            front = 0;
        }
        rear++;
        queue[rear] = value;
        printf("Inserted: %d\n", value);
    }
}


void dequeue()
{
    if(front == -1 || front > rear)
    {
        printf("Underflow\n");
    }
    else
    {
        printf("Deleted: %d\n", queue[front]);
        front++;
    }
}


void display()
{
    if(front == -1 || front > rear)
    {
        printf("Underflow\n");
    }
    else
    {
        for(i = front; i <= rear; i++)
        {
            printf("%d\n", queue[i]);
        }
        printf("\n");
    }
}


int main()
{
    int choice, value;
    while(1)
    {
        printf("Queue Operations:\n");
        printf("1.Enqueue\n");
        printf("2.Dequeue\n");
        printf("3.Display\n");
        printf("4.Exit\n");
        printf("Choice: ");
        scanf("%d", &choice);
        switch(choice)
        {
            case 1:
                printf("Enter value: ");
                scanf("%d", &value);
                enqueue(value); break;
            case 2:
                dequeue(); break;
            case 3:
                display(); break;
            case 4:
                printf("Exiting\n");
                return 0; break;
        }
    }
}
