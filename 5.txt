firstList = []
secondList = []

firstCount = int(input("Enter size of first list: "))
for index in range(firstCount):
    firstList.append(int(input("Enter value: ")))

secondCount = int(input("Enter size of second list: "))
for index in range(secondCount):
    secondList.append(int(input("Enter value: ")))

if len(firstList) == len(secondList):
    print("Same length")
else:
    print("Different length")

if sum(firstList) == sum(secondList):
    print("Same sum")
else:
    print("Different sum")

sharedValues = []

for value in firstList:
    if value in secondList and value not in sharedValues:
        sharedValues.append(value)

if sharedValues:
    print("Common values:", sharedValues)
else:
    print("No common values")
