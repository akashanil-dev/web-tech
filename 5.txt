import java.util.*;

// Interface
interface Schedulable {
    void setSchedule();
}

// Superclass
class Vehicle {
    void calculateFare() {
        System.out.println("Base fare");
    }
}

// Subclass Bus
class Bus extends Vehicle implements Schedulable {
    @Override
    void calculateFare() {
        System.out.println("Bus fare = ₹50");
    }

    public void setSchedule() {
        System.out.println("Bus schedule set");
    }
}

// Subclass Train
class Train extends Vehicle implements Schedulable {
    @Override
    void calculateFare() {
        System.out.println("Train fare = ₹100");
    }

    public void setSchedule() {
        System.out.println("Train schedule set");
    }
}

// Simple Booking class
class Ticket {
    void bookTicket(String passenger) {
        System.out.println("Ticket booked for: " + passenger);
    }
}

// Thread class using Runnable
class BookingThread implements Runnable {
    String passenger;

    BookingThread(String passenger) {
        this.passenger = passenger;
    }

    public void run() {
        System.out.println(passenger + " is booking ticket...");
    }
}

// Main class
public class pg1 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Objects
        Bus bus = new Bus();
        Train train = new Train();

        bus.calculateFare();
        train.calculateFare();
        bus.setSchedule();
        train.setSchedule();

        // Passenger input
        System.out.print("\nEnter number of passengers: ");
        int n = sc.nextInt();
        sc.nextLine();

        String[] passengers = new String[n];
        for (int i = 0; i < n; i++) {
            System.out.print("Enter passenger name: ");
            passengers[i] = sc.nextLine();
        }

        // Search
        System.out.print("\nEnter name to search: ");
        String search = sc.nextLine();
        boolean found = false;
        for (String p : passengers) {
            if (p.equalsIgnoreCase(search)) {
                System.out.println("Passenger found: " + p);
                found = true;
                break;
            }
        }
        if (!found) {
            System.out.println("Passenger not found");
        }

        // Ticket booking
        Ticket ticket = new Ticket();
        System.out.print("\nEnter passenger name for booking: ");
        String name = sc.nextLine();
        ticket.bookTicket(name);

        // Threads
        System.out.println("\n--- Booking Threads ---");
        for (int i = 0; i < n; i++) {
            Thread t = new Thread(new BookingThread(passengers[i]));
            t.start();
        }

        sc.close();
    }
}
