import java.util.*;

// Interface
interface Controllable {
    void control();
}

// Superclass
class Device {
    String name;

    Device(String name) {
        this.name = name;
    }

    void turnOn() {
        System.out.println("Device turned on");
    }
}

// Subclass Light
class Light extends Device implements Controllable {
    Light(String name) {
        super(name);
    }

    @Override
    void turnOn() {
        System.out.println(name + " Light is ON");
    }

    public void control() {
        System.out.println(name + " Light is being controlled");
    }
}

// Subclass Fan
class Fan extends Device implements Controllable {
    Fan(String name) {
        super(name);
    }

    @Override
    void turnOn() {
        System.out.println(name + " Fan is ON");
    }

    public void control() {
        System.out.println(name + " Fan is being controlled");
    }
}

// Simple class instead of inner class (PDF style)
class Settings {
    void applySettings(String device) {
        System.out.println("Settings applied to: " + device);
    }
}

// Thread class
class DeviceThread implements Runnable {
    String device;

    DeviceThread(String device) {
        this.device = device;
    }

    public void run() {
        System.out.println(device + " is being operated...");
    }
}

// Main class
public class DeviceSystem {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Device objects
        System.out.print("Enter light name: ");
        String lname = sc.nextLine();
        Light light = new Light(lname);
        light.turnOn();
        light.control();

        System.out.print("\nEnter fan name: ");
        String fname = sc.nextLine();
        Fan fan = new Fan(fname);
        fan.turnOn();
        fan.control();

        // Array of device names
        System.out.print("\nEnter number of devices: ");
        int n = sc.nextInt();
        sc.nextLine();

        String[] devices = new String[n];
        for (int i = 0; i < n; i++) {
            System.out.print("Enter device name: ");
            devices[i] = sc.nextLine();
        }

        // Settings (normal class)
        Settings s = new Settings();
        System.out.print("\nEnter device name to apply settings: ");
        String dname = sc.nextLine();
        s.applySettings(dname);

        // Threads
        System.out.println("\n--- Device Control Threads ---");
        for (int i = 0; i < n; i++) {
            Thread t = new Thread(new DeviceThread(devices[i]));
            t.start();
        }

        sc.close();
    }
}
