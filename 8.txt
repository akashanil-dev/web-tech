import java.util.*;

// Interface
interface Billable {
    void generateBill();
}

// Superclass
class Person {
    String name;

    Person(String name) {
        this.name = name;
    }

    void getDetails() {
        System.out.println("Name: " + name);
    }
}

// Subclass Doctor
class Doctor extends Person implements Billable {
    Doctor(String name) {
        super(name);
    }

    @Override
    void getDetails() {
        System.out.println("Doctor Name: " + name);
    }

    public void generateBill() {
        System.out.println("Doctor consultation bill: ₹500");
    }
}

// Subclass Patient
class Patient extends Person implements Billable {
    Patient(String name) {
        super(name);
    }

    @Override
    void getDetails() {
        System.out.println("Patient Name: " + name);
    }

    public void generateBill() {
        System.out.println("Patient bill generated");
    }
}

// Thread class
class AppointmentThread implements Runnable {
    String patientName;

    AppointmentThread(String patientName) {
        this.patientName = patientName;
    }

    public void run() {
        System.out.println(patientName + " booked an appointment.");
    }
}

// Main class
public class HospitalSystem {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Doctor input
        System.out.print("Enter doctor name: ");
        String dname = sc.nextLine();
        Doctor doctor = new Doctor(dname);
        doctor.getDetails();
        doctor.generateBill();

        // Patient records (array)
        System.out.print("\nEnter number of patients: ");
        int n = sc.nextInt();
        sc.nextLine(); // Clear buffer

        String[] patients = new String[n];
        for (int i = 0; i < n; i++) {
            System.out.print("Enter patient name: ");
            patients[i] = sc.nextLine();
        }

        // Search patient
        System.out.print("\nEnter patient name to search: ");
        String search = sc.nextLine();
        boolean found = false;
        for (String p : patients) {
            if (p.equalsIgnoreCase(search)) {
                System.out.println("Patient found: " + p);
                found = true;
                break;
            }
        }
        if (!found) {
            System.out.println("Patient not found");
        }

        // Single patient object for billing
        System.out.print("\nEnter a patient name for billing: ");
        String pname = sc.nextLine();
        Patient patient = new Patient(pname);
        patient.getDetails();
        patient.generateBill();

        // Threads for appointments
        System.out.println("\n--- Appointment Booking ---");
        for (int i = 0; i < n; i++) {
            Thread t = new Thread(new AppointmentThread(patients[i]));
            t.start();
        }

        sc.close();
    }
}
