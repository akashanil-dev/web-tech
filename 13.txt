import java.util.*;

// Interface
interface Payable {
    void makePayment();
}

// Superclass
class Policy {
    String id;

    Policy(String id) {
        this.id = id;
    }

    void processClaim() {
        System.out.println("Processing claim...");
    }
}

// Subclass HealthPolicy
class HealthPolicy extends Policy implements Payable {
    HealthPolicy(String id) {
        super(id);
    }

    @Override
    void processClaim() {
        System.out.println("Health policy claim processed for ID: " + id);
    }

    public void makePayment() {
        System.out.println("Health policy payment done");
    }
}

// Subclass CarPolicy
class CarPolicy extends Policy implements Payable {
    CarPolicy(String id) {
        super(id);
    }

    @Override
    void processClaim() {
        System.out.println("Car policy claim processed for ID: " + id);
    }

    public void makePayment() {
        System.out.println("Car policy payment done");
    }
}

// Simple class instead of static inner class (PDF style)
class Agent {
    void assignAgent(String name) {
        System.out.println("Agent assigned: " + name);
    }
}

// Thread class
class ClaimThread implements Runnable {
    String customer;

    ClaimThread(String customer) {
        this.customer = customer;
    }

    public void run() {
        System.out.println(customer + " is filing a claim...");
    }
}

// Main class
public class PolicySystem {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Policy objects
        System.out.print("Enter health policy ID: ");
        String hid = sc.nextLine();
        HealthPolicy hp = new HealthPolicy(hid);
        hp.processClaim();
        hp.makePayment();

        System.out.print("\nEnter car policy ID: ");
        String cid = sc.nextLine();
        CarPolicy cp = new CarPolicy(cid);
        cp.processClaim();
        cp.makePayment();

        // Array of policy IDs
        System.out.print("\nEnter number of policies: ");
        int n = sc.nextInt();
        sc.nextLine();

        String[] policies = new String[n];
        for (int i = 0; i < n; i++) {
            System.out.print("Enter policy ID: ");
            policies[i] = sc.nextLine();
        }

        // Agent (normal class)
        Agent a = new Agent();
        System.out.print("\nEnter agent name: ");
        String aname = sc.nextLine();
        a.assignAgent(aname);

        // Threads
        System.out.println("\n--- Claim Threads ---");
        for (int i = 0; i < n; i++) {
            Thread t = new Thread(new ClaimThread("Customer " + (i + 1)));
            t.start();
        }

        sc.close();
    }
}
