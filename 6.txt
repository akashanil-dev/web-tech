import java.util.*;

// Interface
interface Evaluatable {
    void evaluatePerformance();
}

// Superclass
class Person {
    String name;

    Person(String name) {
        this.name = name;
    }

    void getDetails() {
        System.out.println("Name: " + name);
    }
}

// Subclass Student
class Student extends Person implements Evaluatable {
    int[] marks;

    Student(String name, int[] marks) {
        super(name);
        this.marks = marks;
    }

    @Override
    void getDetails() {
        System.out.println("Student Name: " + name);
    }

    public void evaluatePerformance() {
        int sum = 0;
        for (int m : marks) {
            sum += m;
        }
        double avg = (double) sum / marks.length;
        System.out.println("Average Marks: " + avg);
    }
}

// Subclass Faculty
class Faculty extends Person implements Evaluatable {
    Faculty(String name) {
        super(name);
    }

    @Override
    void getDetails() {
        System.out.println("Faculty Name: " + name);
    }

    public void evaluatePerformance() {
        System.out.println("Faculty performance evaluated based on teaching.");
    }
}

// Thread class
class SubmissionThread implements Runnable {
    String studentName;

    SubmissionThread(String studentName) {
        this.studentName = studentName;
    }

    public void run() {
        System.out.println(studentName + " submitted assignment.");
    }
}

// Main class
public class pg2 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Student input
        System.out.print("Enter student name: ");
        String sname = sc.nextLine();
        System.out.print("Enter number of subjects: ");
        int n = sc.nextInt();
        int[] marks = new int[n];
        for (int i = 0; i < n; i++) {
            System.out.print("Enter mark " + (i + 1) + ": ");
            marks[i] = sc.nextInt();
        }
        sc.nextLine(); // clear buffer

        Student student = new Student(sname, marks);
        student.getDetails();
        student.evaluatePerformance();

        // Faculty input
        System.out.print("\nEnter faculty name: ");
        String fname = sc.nextLine();
        Faculty faculty = new Faculty(fname);
        faculty.getDetails();
        faculty.evaluatePerformance();

        // Multiple students submission (Threads)
        System.out.print("\nEnter number of students submitting assignments: ");
        int count = sc.nextInt();
        sc.nextLine();

        String[] students = new String[count];
        for (int i = 0; i < count; i++) {
            System.out.print("Enter student name: ");
            students[i] = sc.nextLine();
        }

        System.out.println("\n--- Assignment Submissions ---");
        for (int i = 0; i < count; i++) {
            Thread t = new Thread(new SubmissionThread(students[i]));
            t.start();
        }

        sc.close();
    }
}
