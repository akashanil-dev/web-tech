<script>
// ---------------- DATA SOURCE ----------------
const LOCATION_DATA = {
  India: {
    Kerala: ["Alappuzha","Thrissur","Kozhikode"],
    Karnataka: ["Bengaluru","Mysuru","Mangaluru"]
  },
  Canada: {
    Ontario: ["Toronto","Ottawa","Hamilton"],
    Alberta: ["Calgary","Edmonton","Red Deer"]
  }
};

// ---------------- INIT ----------------
$(function(){

  // Populate countries
  $.each(LOCATION_DATA, function(country){
    $("#countrySel").append(`<option value="${country}">${country}</option>`);
  });

  // Country → State
  $("#countrySel").on("change", function(){
    const country = $(this).val();
    $("#stateSel").html(`<option value="">Choose state</option>`);
    $("#districtSel").html(`<option value="">Choose district</option>`);

    if(country){
      $.each(LOCATION_DATA[country], function(state){
        $("#stateSel").append(`<option value="${state}">${state}</option>`);
      });
    }
  });

  // State → District
  $("#stateSel").on("change", function(){
    const country = $("#countrySel").val();
    const state = $(this).val();
    $("#districtSel").html(`<option value="">Choose district</option>`);

    if(state){
      $.each(LOCATION_DATA[country][state], function(_, dist){
        $("#districtSel").append(`<option value="${dist}">${dist}</option>`);
      });
    }
  });

  // ---------------- VALIDATION ----------------
  $("#regForm").on("submit", function(e){
    e.preventDefault();
    let ok = true;
    $(".error").text("");
    $("#finalMsg").text("");

    if($("#fullName").val().trim() === ""){
      $("#errName").text("Name cannot be empty");
      ok = false;
    }

    const mail = $("#mail").val().trim();
    if(mail === "" || !mail.includes("@")){
      $("#errMail").text("Invalid email format");
      ok = false;
    }

    if($("#countrySel").val() === ""){
      $("#errCountry").text("Select a country");
      ok = false;
    }

    if($("#stateSel").val() === ""){
      $("#errState").text("Select a state");
      ok = false;
    }

    if($("#districtSel").val() === ""){
      $("#errDistrict").text("Select a district");
      ok = false;
    }

    if(ok){
      $("#finalMsg").text("Registration completed successfully!");
      $("#regForm")[0].reset();
    }
  });

});
</script></body></html>
