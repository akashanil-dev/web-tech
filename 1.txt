#!/bin/bash

# Configuration
URL="http://129.159.228.210:8000"
USERNAME="the_user"
PASSWORD="jj7*7hggfFHghjtGJh%GhfgfjyFJH%456"

# 1. Login and extract token (requires 'jq')
echo "Logging in..."
TOKEN=$(curl -s -X POST "$URL/login" \
     -H "Content-Type: application/json" \
     -d "{\"username\": \"$USERNAME\", \"password\": \"$PASSWORD\"}" | jq -r '.access_token')

if [ "$TOKEN" == "null" ]; then
    echo "Login failed. Check your server or credentials."
    exit 1
fi

echo "Connected! Type your message (Ctrl+C to quit)"

# 2. Interactive Loop
while true; do
    echo -n "You > "
    read USER_INPUT
    
    # Send request and parse reply
    RESPONSE=$(curl -s -X POST "$URL/chat" \
         -H "Authorization: Bearer $TOKEN" \
         -H "Content-Type: application/json" \
         -d "{\"message\": \"$USER_INPUT\"}" | jq -r '.reply')
    
    echo -e "AI  > $RESPONSE\n"
done
