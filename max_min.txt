#include <stdio.h>
#define MAX 50
int a[MAX];

void MaxMin(int i, int j, int *max, int *min)
{
    int mid;
    int max1, min1;
    if (i == j)
    {
        *max = *min = a[i];
    }
    else if (i == j - 1)
    {
        if (a[i] < a[j])
        {
            *max = a[j];
            *min = a[i];
        }
        else
        {
            *max = a[i];
            *min = a[j];
        }
    }
    else
    {
        mid = (i + j) / 2;
        MaxMin(i, mid, max, min);
        MaxMin(mid + 1, j, &max1, &min1);
        if (*max < max1)
            *max = max1;
        if (*min > min1)
            *min = min1;
    }
}

int main()
{
    int n, i;
    int max, min;
    printf("Enter number of elements (<= %d): ", MAX);
    scanf("%d", &n);
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++)
    {
        scanf("%d", &a[i]);
    }
    MaxMin(0, n - 1, &max, &min);
    printf("Maximum = %d\n", max);
    printf("Minimum = %d\n", min);
    return 0;
}
