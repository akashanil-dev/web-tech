import java.util.Scanner;
abstract class Staff {
    String name;
    abstract void display();
}
class InvalidSalaryException extends Exception {
    public InvalidSalaryException(String msg) {
        super(msg);
    }
}
class Employee extends Staff {
    private int empId;
    protected String designation;
    protected double basicSalary;
    double totalSalary;

    Employee() {}

    Employee(int id, String name, String desig, double salary) throws InvalidSalaryException {
        if (salary < 0) throw new InvalidSalaryException("Invalid Salary");
        this.empId = id;
        this.name = name;
        this.designation = desig;
        this.basicSalary = salary;
    }

    double calculateSalary(double bonus) {
        totalSalary = basicSalary + bonus;
        return totalSalary;
    }

    double calculateSalary(double bonus, double overtime) {
        totalSalary = basicSalary + bonus + overtime;
        return totalSalary;
    }

    class Payroll {
        String slip;

        void generateSlip() {
            slip = "Employee: " + name + 
                   ", Designation: " + designation + 
                   ", Salary: " + totalSalary;
        }

        String getSlip() {
            return slip;
        }
    }

    public void display() {
        System.out.println("ID: " + empId + ", Name: " + name + ", Designation: " + designation);
    }
}
class Manager extends Employee {

    Manager(int id, String name, String desig, double salary) throws InvalidSalaryException {
        super(id, name, desig, salary);
    }

    @Override
    double calculateSalary(double bonus, double overtime) {
        double extra = 1000; 
        totalSalary = basicSalary + bonus + overtime + extra;
        return totalSalary;
    }
}
class Developer extends Employee {

    Developer(int id, String name, String desig, double salary) throws InvalidSalaryException {
        super(id, name, desig, salary);
    }

    @Override
    double calculateSalary(double bonus, double overtime) {
        double extra = 500; 
        totalSalary = basicSalary + bonus + overtime + extra;
        return totalSalary;
    }
}
class SalaryThread extends Thread {
    public void run() {
        System.out.println("Processing salary...");
    }
}
class ReportThread extends Thread {
    String data;

    ReportThread(String data) {
        this.data = data;
    }

    public void run() {
        try {
            java.io.FileWriter fw = new java.io.FileWriter("payroll.txt", true);
            fw.write(data + "\n");
            fw.close();
            System.out.println("Report saved.");
        } catch (Exception e) {
            System.out.println(e);
        }
    }
}
public class MainApp {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        try {
            System.out.print("Enter Employee ID: ");
            int id = sc.nextInt();
            sc.nextLine();

            System.out.print("Enter Name: ");
            String name = sc.nextLine();

            System.out.print("Enter Designation: ");
            String desig = sc.nextLine();

            System.out.print("Enter Basic Salary: ");
            double salary = sc.nextDouble();

            System.out.print("Type (1-Manager, 2-Developer): ");
            int type = sc.nextInt();

            Employee emp;

            if (type == 1)
                emp = new Manager(id, name, desig, salary);
            else
                emp = new Developer(id, name, desig, salary);

            System.out.print("Enter Bonus: ");
            double bonus = sc.nextDouble();

            System.out.print("Enter Overtime: ");
            double overtime = sc.nextDouble();

            double total = emp.calculateSalary(bonus, overtime);

            if (total < 0) throw new InvalidSalaryException("Invalid Total Salary");

            Employee.Payroll p = emp.new Payroll();
            p.generateSlip();

            emp.display();
            System.out.println("Salary Slip: " + p.getSlip());

            SalaryThread t1 = new SalaryThread();
            ReportThread t2 = new ReportThread(p.getSlip());

            t1.start();
            t2.start();

        } catch (InvalidSalaryException e) {
            System.out.println(e.getMessage());
        }
    }
}
