#include <stdio.h>

void main()
{
    int choice, size, numElements, deleteValue, i, j, found;

    printf("Enter the size of the array: ");
    scanf("%d", &size);

    int numbers[size];

    printf("Enter %d elements: ", size);
    for (i = 0; i < size; i++)
    {
        scanf("%d", &numbers[i]);
    }

    printf("\nChoose an operation:\n");
    printf("1. Insert elements\n");
    printf("2. Delete an element\n");
    printf("3. Traverse array\n");
    printf("4. Exit\n");
    printf("Enter your choice: ");
    scanf("%d", &choice);

    switch (choice)
    {
        case 1:
            printf("How many elements do you want to insert? ");
            scanf("%d", &numElements);

            for (i = 0; i < numElements; i++)
            {
                int newValue;
                printf("Enter element %d: ", i + 1);
                scanf("%d", &newValue);
                numbers[size] = newValue;
                size++;
            }
            printf("\nUpdated array: ");
            for (i = 0; i < size; i++)
                printf("%d ", numbers[i]);
            break;

        case 2:
            printf("Enter the value to delete: ");
            scanf("%d", &deleteValue);
            found = 0;

            for (i = 0; i < size; i++)
            {
                if (numbers[i] == deleteValue)
                {
                    for (j = i; j < size - 1; j++)
                        numbers[j] = numbers[j + 1];
                    size--;
                    found = 1;
                    break;
                }
            }

            if (found)
            {
                printf("\nElement deleted successfully.\n");
                printf("Updated array: ");
                for (i = 0; i < size; i++)
                    printf("%d ", numbers[i]);
            }
            else
                printf("\nElement not found in array.\n");

            break;

        case 3:
            printf("\nArray elements: ");
            for (i = 0; i < size; i++)
                printf("%d ", numbers[i]);
            break;

        case 4:
            printf("Exiting program...\n");
            break;

        default:
            printf("Invalid choice!\n");
    }
}
