#include <stdio.h>

int main()
{
    int limit, a = 0, b = 1, result, i;

    printf("Enter the limit: ");
    scanf("%d", &limit);

    printf("Fibonacci Series up to %d terms:\n", limit);

    for (i = 1; i <= limit; i++)
    {
        printf("%d ", a);
        result = a + b;
        a = b;
        b = result;
    }

    printf("\n");
    return 0;
}
