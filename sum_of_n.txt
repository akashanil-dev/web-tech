//To Implement a program to find sum of n natural numbers using while loop
#include <stdio.h>
void main()
{
    int sum = 0 ;
    int i = 0;
    int count;
    printf("Enter the count: ");
    scanf("%d", &count);
    
    while(i <= count)
    {
        sum = sum + i;
        i++;
    }
    printf("The sum is %d", sum);
}
