#include <stdio.h>

int main()
{
    int input, original, remainder, digits = 0;
    int result = 0;

    printf("Enter a number: ");
    scanf("%d", &input);

    original = input;

    int temp = input;
    while (temp != 0
    {
        temp /= 10;
        digits++;
    }

    temp = input;
    while (temp != 0)
    {
        remainder = temp % 10;

        int power = 1;
        for (int i = 0; i < digits; i++)
	        {
            power *= remainder;  // multiply remainder 'digits' times
        }

        result += power;
        temp /= 10;
    }

    if (result == original)
        printf("%d is an Armstrong number.\n", original);
    else
        printf("%d is not an Armstrong number.\n", original);

    return 0;
}
