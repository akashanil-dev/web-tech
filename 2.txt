@echo off
setlocal EnableDelayedExpansion

:: =====================================
:: Configuration
:: =====================================
set "URL=http://129.159.228.210:8000"
set "USERNAME=the_user"

:: Escape % signs as %%
set "PASSWORD=jj7*7hggfFHghjtGJh%%GhfgfjyFJH%%456"

echo Logging in...
echo.

:: =====================================
:: Login
:: =====================================
curl -s -X POST "%URL%/login" ^
-H "Content-Type: application/json" ^
-d "{\"username\":\"%USERNAME%\",\"password\":\"%PASSWORD%\"}" > login.json

:: Use PowerShell to extract token properly
for /f "delims=" %%a in ('
    powershell -Command ^
    "(Get-Content login.json | ConvertFrom-Json).access_token"
') do (
    set "TOKEN=%%a"
)

if "!TOKEN!"=="" (
    echo Login failed.
    echo.
    type login.json
    echo.
    pause
    exit /b
)

echo Connected! Type your message. Commands: /clear, exit
echo.

:: =====================================
:: Chat Loop
:: =====================================
:loop

set "USER_INPUT="
set /p USER_INPUT=You: 

if /I "!USER_INPUT!"=="exit" (
    goto end
)

:: =====================================
:: Clear history
:: =====================================
if /I "!USER_INPUT!"=="/clear" (

    curl -s -X POST "%URL%/clear" ^
    -H "Authorization: Bearer !TOKEN!" > nul

    echo.
    echo AI  ^> [History Cleared]
    echo.

    goto loop
)

:: =====================================
:: Send message
:: =====================================
curl -s -X POST "%URL%/chat" ^
-H "Authorization: Bearer !TOKEN!" ^
-H "Content-Type: application/json" ^
-d "{\"message\":\"!USER_INPUT!\"}" > response.json

:: =====================================
:: Display AI reply properly
:: =====================================
echo.
echo AI  ^>
echo.

powershell -Command ^
"$r = Get-Content response.json -Raw | ConvertFrom-Json; Write-Output $r.reply"

echo.
goto loop

:: =====================================
:: Cleanup
:: =====================================
:end

if exist login.json del login.json
if exist response.json del response.json

echo.
echo Goodbye.
pause
