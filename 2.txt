@echo off
setlocal

:: ================================
:: Configuration
:: ================================
set "URL=http://129.159.228.210:8000"
set "USERNAME=the_user"

:: IMPORTANT:
:: Every % sign inside batch files must become %%
set "PASSWORD=jj7*7hggfFHghjtGJh%%GhfgfjyFJH%%456"

echo Logging in...
echo.

:: ================================
:: Login and get token
:: ================================
curl -s -X POST "%URL%/login" ^
-H "Content-Type: application/json" ^
-d "{\"username\":\"%USERNAME%\",\"password\":\"%PASSWORD%\"}" > login.json

:: ================================
:: Extract token
:: ================================
set TOKEN=

for /f "tokens=2 delims=:,}" %%a in ('findstr "access_token" login.json') do (
    set TOKEN=%%a
)

:: Remove quotes and spaces
set TOKEN=%TOKEN:"=%
set TOKEN=%TOKEN: =%

:: ================================
:: Login failed check
:: ================================
if "%TOKEN%"=="" (
    echo Login failed.
    echo.
    echo Possible reasons:
    echo - Server is offline
    echo - Wrong IP address
    echo - Wrong username/password
    echo - curl is missing
    echo.

    type login.json
    echo.

    pause
    exit /b
)

echo Connected successfully.
echo.
echo Commands:
echo   /clear   - Clear chat history
echo   exit     - Quit
echo.

:: ================================
:: Main chat loop
:: ================================
:loop

set "USER_INPUT="
set /p USER_INPUT=You ^> 

:: Exit command
if /I "%USER_INPUT%"=="exit" (
    goto end
)

:: ================================
:: Clear history
:: ================================
if /I "%USER_INPUT%"=="/clear" (

    curl -s -X POST "%URL%/clear" ^
    -H "Authorization: Bearer %TOKEN%" > nul

    echo.
    echo AI  ^> [History Cleared]
    echo.

    goto loop
)

:: ================================
:: Send message
:: ================================
curl -s -X POST "%URL%/chat" ^
-H "Authorization: Bearer %TOKEN%" ^
-H "Content-Type: application/json" ^
-d "{\"message\":\"%USER_INPUT%\"}" > response.json

:: ================================
:: Extract AI reply
:: ================================
set REPLY=

for /f "tokens=2 delims=:}" %%a in ('findstr "reply" response.json') do (
    set REPLY=%%a
)

:: Cleanup formatting
set REPLY=%REPLY:"=%
set REPLY=%REPLY:\n=%
set REPLY=%REPLY:~1%

echo.
echo AI  ^> %REPLY%
echo.

goto loop

:: ================================
:: Exit cleanup
:: ================================
:end

if exist login.json del login.json
if exist response.json del response.json

echo.
echo Goodbye.
pause
