@echo off
setlocal enabledelayedexpansion

:: Configuration - Update with your Oracle Server IP
set URL=http://129.159.228.210:8000
set USERNAME=the_user
set PASSWORD=jj7*7hggfFHghjtGJh%GhfgfjyFJH%456

echo Logging in...

:: 1. Login to get Token (Using curl and a bit of batch magic to parse JSON)
for /f "tokens=2 delims=:, " %%a in ('curl -s -X POST %URL%/login -H "Content-Type: application/json" -d "{\"username\":\"%USERNAME%\",\"password\":\"%PASSWORD%\"}"') do (
    set TOKEN=%%a
)

:: Clean the token (remove quotes)
set TOKEN=%TOKEN:"=%
set TOKEN=%TOKEN:} =%

if "%TOKEN%"=="" (
    echo Login failed! Check your server IP or credentials.
    pause
    exit /b
)

echo Connected! Type your message. 
echo Type /clear to reset history, or Ctrl+C to quit.
echo.

:loop
set /p USER_INPUT="You > "

:: Check for /clear command
if "!USER_INPUT!"=="/clear" (
    curl -s -X POST %URL%/clear -H "Authorization: Bearer %TOKEN%" > nul
    echo AI  ^> [History Cleared]
    echo.
    goto loop
)

:: 2. Send Chat Request
:: Note: We use a temp file because CMD struggles with complex JSON characters in strings
curl -s -X POST %URL%/chat ^
     -H "Authorization: Bearer %TOKEN%" ^
     -H "Content-Type: application/json" ^
     -d "{\"message\":\"!USER_INPUT!\"}" > response.json

:: Simple way to show the reply from the JSON file
echo AI  ^> 
type response.json | findstr /v "{" | findstr /v "}"
echo.

goto loop
