import java.util.Scanner;
abstract class BankUser {
    String name;

    abstract void displayUser();
}
class InsufficientBalanceException extends Exception {
    public InsufficientBalanceException(String msg) {
        super(msg);
    }
}
class Account extends BankUser {
    int accNo;
    double balance;

    
    Account() {
        accNo = 0;
        name = "Unknown";
        balance = 0;
    }

    Account(int accNo, String name, double balance) {
        this.accNo = accNo;
        this.name = name;
        this.balance = balance;
    }

    
    void deposit(int amount) {
        balance += amount;
    }

    void deposit(double amount) {
        balance += amount;
    }

    void withdraw(double amount) throws InsufficientBalanceException {
        if (amount > balance)
            throw new InsufficientBalanceException("Insufficient Balance");
        balance -= amount;
    }

    public void displayUser() {
        System.out.println("AccNo: " + accNo + ", Name: " + name + ", Balance: " + balance);
    }

    static class BankInfo {
        static void showBank() {
            System.out.println("ABC Bank");
        }
    }

    class Transaction {
        String type;
        double amount;

        Transaction(String type, double amount) {
            this.type = type;
            this.amount = amount;
        }

        String getDetails() {
            return "Transaction: " + type + " Amount: " + amount;
        }
    }
}
class SavingsAccount extends Account {

    SavingsAccount(int accNo, String name, double balance) {
        super(accNo, name, balance);
    }

    @Override
    void withdraw(double amount) throws InsufficientBalanceException {
        if (balance - amount < 500)
            throw new InsufficientBalanceException("Minimum balance required 500");
        balance -= amount;
    }
}
class CurrentAccount extends Account {

    CurrentAccount(int accNo, String name, double balance) {
        super(accNo, name, balance);
    }

    @Override
    void withdraw(double amount) throws InsufficientBalanceException {
        if (amount > balance + 1000)
            throw new InsufficientBalanceException("Overdraft limit exceeded");
        balance -= amount;
    }
}
class BalanceThread extends Thread {
    Account acc;

    BalanceThread(Account acc) {
        this.acc = acc;
    }

    public void run() {
        System.out.println("Checking Balance: " + acc.balance);
    }
}
class TransactionThread extends Thread {
    String data;

    TransactionThread(String data) {
        this.data = data;
    }

    public void run() {
        try {
            java.io.FileWriter fw = new java.io.FileWriter("transactions.txt", true);
            fw.write(data + "\n");
            fw.close();
            System.out.println("Transaction saved.");
        } catch (Exception e) {
            System.out.println(e);
        }
    }
}

public class MainApp {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        try {
            System.out.print("Enter Account Number: ");
            int accNo = sc.nextInt();
            sc.nextLine();

            System.out.print("Enter Name: ");
            String name = sc.nextLine();

            System.out.print("Enter Initial Balance: ");
            double balance = sc.nextDouble();

            System.out.print("Enter Account Type (1-Savings, 2-Current): ");
            int type = sc.nextInt();

            Account acc;

            if (type == 1)
                acc = new SavingsAccount(accNo, name, balance);
            else
                acc = new CurrentAccount(accNo, name, balance);

            System.out.print("Enter deposit amount: ");
            double dep = sc.nextDouble();
            acc.deposit(dep);

            System.out.print("Enter withdraw amount: ");
            double wd = sc.nextDouble();
            acc.withdraw(wd);

            acc.displayUser();

            String summary = "Transaction Done for " + name;

            BalanceThread t1 = new BalanceThread(acc);
            TransactionThread t2 = new TransactionThread(summary);

            t1.start();
            t2.start();

        } catch (InsufficientBalanceException e) {
            System.out.println(e.getMessage());
        }
    }
}
