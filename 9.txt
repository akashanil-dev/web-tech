insert into customer (cust_id,name,city,branch,deposit) values
(1,'rahul','nagpur','nagpur',60000),
(2,'neha','bombay','bombay',4000),
(3,'arjun','pune','nagpur',5500),
(4,'meera','nagpur','nagpur',30000),
(5,'vishal','bombay','bombay',7000),
(6,'kiran','delhi','delhi',50000),
(7,'anita','kochi','kochi',20000),
(8,'sanjay','kochi','kochi',15000);

select branch,sum(deposit) as amount from customer group by branch having(sum(deposit)>50000);

select branch,sum(deposit) as amount from customer where city="bombay" group by branch having(sum(deposit)>5000);

select name from customer group by name,deposit having deposit=(select max(deposit) from customer);

select branch,sum(deposit) as total_deposit from customer group by branch having sum(deposit) >= all (select sum(deposit) from customer group by branch);
