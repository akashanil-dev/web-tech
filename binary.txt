#include<stdio.h>
int main()
{
    int n, x, low, high, mid, i;
    printf("Enter Number Of Elements: ");
    scanf("%d", &n);
    int arr[n];
    printf("Enter %d Sorted Elements:\n", n);
    for (i = 0; i < n; i++)
    {
        printf("Element %d: ", i + 1);
        scanf("%d", &arr[i]);
    }
    printf("Search Value: ");
    scanf("%d", &x);
    low = 0;
    high = n - 1;
    int found = 0;
    while (low <= high)
    {
        mid = (low + high) / 2;
        if (arr[mid] == x)
        {
            printf("Found At Index %d\n", mid);
            found = 1; break;
        }
        else if (arr[mid] < x)
            low = mid + 1;
        else
            high = mid - 1;
    }
    if (!found)
        printf("Not Found\n");
    return 0;
}
